#!/usr/bin/env python3
"""Build the P3 per-lever attribution table from attribution-raw.json (p3_attribution.py.txt),
the stock analyzer's c1/ c2/ output (tools/decode_timeline.py) and the workstream projections.

  cd results/2026-09-26-serve-r3/P3 && python3 p3_table.py.txt   -> attribution.json

Every measured value is [rank0, rank1] from one profiled window per arm (A_p-1, K_p-1). The
windows are c=1: one L.A.I.L request, 200 tokens (A 98 / K 105 decode steps at m=4), and c=2:
one pair of 80-token streams (A 30 / K 34 steps at m=8). Profiled steps run slower than
unprofiled ones, and host-side spans are inflated the most (fusion-host iterations.txt 0: about
1.7x on the stock Engram stage).
"""
import json
import statistics as st

R = json.load(open("attribution-raw.json"))
W = ("c1", "c2")


def g(arm, w, r):
    return R[f"{arm}_{w}_r{r}"]


def pair(fn, w):
    return [round(fn(w, r), 3) for r in (0, 1)]


def d(fa, fk, w):
    return [round(fk(w, r) - fa(w, r), 3) for r in (0, 1)]


def cat(arm, name):
    return lambda w, r: g(arm, w, r)["categories"].get(name, {"ms_per_step": 0.0})["ms_per_step"]


def seg(arm, name):
    return lambda w, r: g(arm, w, r)["layer_segments"][name]["ms_per_step_mean_x40"]


def ffn(arm, name, key="median_us"):
    return lambda w, r: g(arm, w, r)["ffn_prologue_and_mhc"][name][key]


def eb(arm, name, key="median"):
    return lambda w, r: g(arm, w, r)["eager_before_next_target"][name][key]


def roles_A(w, r):
    j = json.load(open(f"A_p-1/{w}/{w}-rank{r}.json"))
    grp = max(j["groups"].values(), key=lambda x: x["steps"])
    return {x["role"]: x for x in grp["gemm_roles"]}


def x40(f):
    return lambda w, r: f(w, r) * 40 / 1e3


out = {"what": "ARMS-r3 section 6 (P3): per-lever attribution by kernel and by critical path, A_p (round-34 "
               "defaults) vs K_p (defaults + the 10 promoted envs), both ranks, c=1 L.A.I.L window and c=2 pair window",
       "boots": {"A_p-1": "results/2026-09-26-serve-r3/P3/A_p-1 (HEAD 54ad48e, review-e14, gpu_guard rc 0)",
                 "K_p-1": "results/2026-09-26-serve-r3/P3/K_p-1 (HEAD 66d5960, same docker/patch sha256, review-e14, gpu_guard rc 0)"},
       "units": "ms per decode step unless named; every value is [rank0, rank1]",
       "windows": {w: {"A_steps": g("A", w, 0)["steps"], "K_steps": g("K", w, 0)["steps"],
                       "A_graph": g("A", w, 0)["target_graph_id"], "K_graph": g("K", w, 0)["target_graph_id"]} for w in W}}

# ---- step level ------------------------------------------------------------------------------------
step = {}
for w in W:
    s = {}
    for k in ("step_wall_ms", "device_busy_ms", "device_idle_ms", "device_ops_per_step"):
        s[k] = {"A": pair(lambda w, r: g("A", w, r)[k]["median"], w), "K": pair(lambda w, r: g("K", w, r)[k]["median"], w)}
        s[k]["K_minus_A"] = [round(b - a, 3) for a, b in zip(s[k]["A"], s[k]["K"])]
    for k in ("target_graph_ms", "draft_graph_ms", "eager_after_target_ms", "eager_before_next_target_ms"):
        f = lambda arm: (lambda w, r: g(arm, w, r)["phases_ms"][k]["median"])
        s["phase " + k] = {"A": pair(f("A"), w), "K": pair(f("K"), w), "K_minus_A": d(f("A"), f("K"), w)}
    for k in ("A attn block: prev MoE AR end -> attn AR start", "B attn AR", "C ffn prologue: attn AR end -> routed start",
              "D routed MoE", "E epilogue: routed end -> MoE AR start", "F MoE AR"):
        s["segment " + k] = {"A": pair(seg("A", k), w), "K": pair(seg("K", k), w), "K_minus_A": d(seg("A", k), seg("K", k), w)}
    step[w] = s
out["step_level_profiled"] = step
out["step_level_note"] = ("Phases and segments are medians/means of different sets and do not add exactly. Segments "
                          "A-F are per-layer means x 40 inside the target graph.")

# ---- device ms/step per kernel category --------------------------------------------------------------
cats = {}
for w in W:
    names = set()
    for arm in "AK":
        for r in (0, 1):
            names |= set(g(arm, w, r)["categories"])
    rows = {}
    for c in names:
        a, k = cat("A", c), cat("K", c)
        rows[c] = {"A": pair(a, w), "K": pair(k, w), "K_minus_A": d(a, k, w),
                   "launches_A_r0": g("A", w, 0)["categories"].get(c, {}).get("launches_per_step", 0),
                   "launches_K_r0": g("K", w, 0)["categories"].get(c, {}).get("launches_per_step", 0)}
    cats[w] = dict(sorted(rows.items(), key=lambda kv: -max(kv[1]["A"][0], kv[1]["K"][0])))
out["device_ms_per_step_by_category"] = cats
out["category_note"] = ("Kernel time summed over all streams in counted decode steps (side streams included, so it is "
                        "not the critical path). [LEVER] marks the round-3 categories; the rest are decode_timeline's "
                        "stock CATEGORIES. dense_gemv = dsv41_gemv::gemv_kernel; wp_gemm = _wp_gemv_kernel or the "
                        "cuBLAS 128x1 bf16 kernel at grid (2,1,1); compare_scalar<long> is split out of torch_eltwise.")

# ---- dense roles -------------------------------------------------------------------------------------
roles = {}
for w in W:
    rr = {}

        A = [roles_A(w, r).get(role) for r in (0, 1)]

            continue

                    "ms_per_step_A": [x["ms_per_step"] for x in A], "ms_per_step_K": [x["ms_per_step"] for x in K],
                    "K_minus_A": [round(k["ms_per_step"] - a["ms_per_step"], 3) for a, k in zip(A, K)],
                    "roofline_us_250GBps": A[0].get("roofline_us")}
    roles[w] = rr
out["dense_gemm_roles"] = roles

roof = {}
for w in W:
    rr, tot = {}, [0.0, 0.0]
    for role in ("qkv_a", "wq_b", "wo_a", "wo_b", "shared_gate_up", "shared_down", "lm_head", "engram_wkv", "main_proj"):
        wb = dt.WEIGHTS[role][0]
        vals = []
        for r in (0, 1):
            x = g("K", w, r)["gemv_roles"].get(role)
            if not x:
                continue
            gap = max(0.0, x["us_median"] - wb / 250e9 * 1e6) * x["calls_per_step"] / 1e3
            tot[r] += gap
            vals.append({"GBps": round(wb / (x["us_median"] * 1e-6) / 1e9, 1),
                         "pct_of_250": round(100 * wb / (x["us_median"] * 1e-6) / 250e9, 1),
                         "gap_to_roofline_ms_per_step": round(gap, 3)})
        if vals:
            rr[role] = vals
    rr["sum_gap_ms_per_step"] = [round(t, 3) for t in tot]
    roof[w] = rr
out["dense_gemv_roofline_K"] = roof

# ---- levers ------------------------------------------------------------------------------------------
def dense_total(arm):
    return lambda w, r: sum(cat(arm, c)(w, r) for c in ("dense_b12x", "woa_einsum", "dense_gemv [DENSE_GEMV]"))


def mhc_elapsed(arm):
    return lambda w, r: (ffn(arm, "mhc elapsed, attention half (first start -> last end)")(w, r)
                         + ffn(arm, "mhc elapsed, ffn half (first start -> last end)")(w, r)) * 40 / 1e3


def join(arm):
    return lambda w, r: g(arm, w, r)["layer_segments"]["qkv_join_wait_us"]["wp layers"]["per_step_ms"]


def routed_call(arm, key):
    return lambda w, r: g(arm, w, r)["layer_segments"]["routed_call_us"][key]


L = []


def lever(env, expected, projected, metric, fa, fk, critical, verdict, why, extra=None):
    row = {"env": env, "expected_in_profile (ARMS-r3 s6)": expected, "projected_ms_per_step": projected,
           "metric": metric}
    for w in W:
        row[w] = {"A": pair(fa, w), "K": pair(fk, w), "K_minus_A": d(fa, fk, w)}
    row["on_critical_path"] = critical
    row["verdict"] = verdict
    row["why"] = why
    if extra:
        row.update(extra)
    L.append(row)
