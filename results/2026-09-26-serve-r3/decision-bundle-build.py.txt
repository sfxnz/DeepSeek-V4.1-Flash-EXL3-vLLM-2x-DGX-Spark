# Composes decision-bundle.json from the output of decision-bundle-compute.py.txt:
#   python3 decision-bundle-compute.py.txt raw.json && python3 decision-bundle-build.py.txt raw.json decision-bundle.json
import json, sys
raw = json.load(open(sys.argv[1]))
C = raw["comparisons"]
P = raw["per_boot"]
r = lambda x, n=3: round(x, n)

K_ENV = ["DSV41_P2B_COOP=2", "DSV41_DENSE_GEMV=1", "DSV41_MHC_DET_SPLITS=16", "DSV41_ENGRAM_NATIVE_STAGE=1",
         "DSV41_ENGRAM_EARLY_HASH=1", "DSV41_ATTN_T2R_DEDUP=1", "DSV41_SWA_META_FUSED=1", "DSV41_MOE_PREP_FUSED=1",
         "DSV41_CANDIDATE_MASK_BOUNDED=1", "DSV41_INDEXER_WP_GEMV=1"]

def row(cmpname, grp, key):
    return C[cmpname][grp][key]

def g(cmpname, key):
    v = row(cmpname, "ms_step", key)
    return v

        "primary_gate": {k: {kk: row(name, "ms_step", k)[kk] for kk in

                               for k, v in C[name]["acceptance"].items()},
        "non_inferiority_ok_vs_larger_spread": {k: v["ok_vs_larger_spread"] for k, v in C[name]["non_inferiority"].items()},
    }

boots_K = ["K-1", "K-2"]; boots_KC = ["KC-1", "KC-2"]; boots_A = ["A-1", "A-2"]
honest = {b: dict(prose_long_c1=f"{P[b]['prose_long_c1_finish']}, post-EOS {P[b]['prose_long_c1_post_eos']}",
                  prose_long_c2=f"{P[b]['prose_long_c2_finish']}, post-EOS {P[b]['prose_long_c2_post_eos']}",
                  lail_finish=P[b]["lail_finish"], serve_env_ranks_match=P[b]["serve_env_ranks_match"],
                  lever_disarmed=P[b]["lever_disarmed"], disarm_scan_rc=P[b]["disarm_scan_rc"],
                  gpu_guard_rc=P[b]["gpu_guard_rc"], contaminated=P[b]["contaminated"],
                  e2e_rc=P[b]["e2e_rc"], correctness=P[b]["correctness"],
                  memavail_after_smoke_spark1_spark2=P[b]["memavail_after_smoke_gib"])
          for b in P}
all_honest = all(P[b]["prose_long_c1_finish"] == "length" and P[b]["prose_long_c1_post_eos"] == 0.0
                 and P[b]["prose_long_c2_finish"] == "length" and P[b]["serve_env_ranks_match"]
                 and not P[b]["lever_disarmed"] and P[b]["disarm_scan_rc"] == [0] and P[b]["gpu_guard_rc"] == [0]
                 and not P[b]["contaminated"] and P[b]["e2e_rc"] == [0] and P[b]["correctness"] == "8/8" for b in P)
q_quick_K = all(P[b]["quality"]["quick"]["passed"] for b in boots_K)
q_quick_KC = all(P[b]["quality"]["quick"]["passed"] for b in boots_KC)
q_full_K2 = P["K-2"]["quality"]["full"]["passed"]

KvA = checks["K_vs_A"]
promote_K = (all(v["gate_pass"] for v in KvA["primary_gate"].values())
             and all(KvA["acceptance_matched"].values())
             and all(KvA["non_inferiority_ok_vs_larger_spread"].values())
             and all_honest and q_quick_K and q_full_K2)
KvKC = checks["K_vs_KC"]
coop_beyond = all(v["gate_pass"] for v in KvKC["primary_gate"].values())
promote = (K_ENV if coop_beyond else [e for e in K_ENV if not e.startswith("DSV41_P2B_COOP")]) if promote_K else []

# ---- K vs A deltas table (what the task asks for)
def d(name, key, grp):
    v = C[name][grp][key]
    return {"A": v["base"], "K": v["cand"], "A_median": v["base_median"], "K_median": v["cand_median"],
            "delta": (-v["improvement"] if grp == "ms_step" else v["improvement"]),
            "delta_pct": (-v["improvement_pct"] if grp == "ms_step" else v["improvement_pct"]),
            "noise": v["noise"], "every_K_boot_beats_A_median": v["every_cand_boot_beats_base_median"]}
deltas = {
    "ms_step": {k: d("K_vs_A", k, "ms_step") for k in C["K_vs_A"]["ms_step"]},
    "tok_s": {k: d("K_vs_A", k, "tok_s") for k in C["K_vs_A"]["tok_s"]},
}

# ---- projection vs measured (section 11)

c1_keys = ["four_prose_c1_ms_step", "prose_long_c1_ms_step", "lail_ms_step", "bench_prose_c1_ms_step", "structured_c1_ms_step"]
c2_keys = ["prose_long_c2_ms_step", "bench_prose_c2_ms_step", "structured_c2_ms_step"]
pvm = {}
for name, pr in proj.items():
    ent = {}
    for conc, keys in (("c1", c1_keys), ("c2", c2_keys)):
        lo, hi = pr[conc]
        meas = {k: C[name]["ms_step"][k]["improvement"] for k in keys}
        ent[conc] = {"projected_saving_ms_step": [lo, hi],
                     "measured_saving_ms_step": meas,
                     "measured_minus_projected_ms_step": {k: [r(v - hi), r(v - lo)] for k, v in meas.items()},
                     "measured_above_projection_top": {k: v > hi for k, v in meas.items()}}
    pvm[name] = ent
