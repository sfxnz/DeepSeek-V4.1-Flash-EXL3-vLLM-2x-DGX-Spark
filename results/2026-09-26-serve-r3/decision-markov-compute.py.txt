def boot(b):
    d = BC.boot(b)
    f = json.load(open(f"{b}/four/four_numbers.json"))
    pl = {c["concurrency"]: c for c in f["prose_long_cells"]}
    bd = BC.bench(b)
    d["acc_prose_long_c2"] = pl[2]["median_run_acceptance_len"]
    d["acc_bench_prose_c2"] = bd["prose_c2"]["median_run_acceptance_len"]
    d["acc_structured_c2"] = bd["structured_c2"]["median_run_acceptance_len"]
    d["acc_lail_runs"] = AD.lail_acc_runs(b)
    d["lail_tok_runs_full"] = [json.loads(m.group(1))["lail_tok_s"]
                               for m in (re.match(r"run=\d+ (\{.*\})$", l.strip()) for l in open(f"{b}/lail10.txt")) if m]
    d["four_partial"] = f.get("partial")
    return d


def mem_avail(path):
    return [line.split()[6] for line in open(path) if line.startswith("Mem:")]


def unset(v):

def validity(b, P):
    arm = ARM_OF[b]
    envs = {n: AD.env_of(b, n) for n in (1, 2)}
    k_ok = all(envs[n].get(k) == v for n in (1, 2) for k, v in AD.K_ENV.items())
    kp_off = all(envs[n].get(k) in (None, "", "0") for n in (1, 2) for k in AD.ADDONS)
    arm_state = {k: [envs[1].get(k), envs[2].get(k)] for k in ARM_ENV[arm]}
    arm_ok = all((unset(envs[n].get(k)) if v is None else envs[n].get(k) == v) for n in (1, 2) for k, v in ARM_ENV[arm].items())
    eng = AD.engagement(b)
    sm_total = eng.get(SM)
    k256, k1024 = eng.get(SM + " k=256"), eng.get(SM + " k=1024")
    absent = lambda c: c is None or c == [0, 0]
    present = lambda c: c is not None and c[0] >= 1 and c[1] >= 1
    if arm == "W":
        sm_ok = absent(sm_total) and absent(k256) and absent(k1024)
    elif arm == "S":
        sm_ok = present(k256) and sm_total == k256 and absent(k1024)
    else:
        sm_ok = present(k1024) and sm_total == k1024 and absent(k256)
    k_eng_ok = all(present(eng.get(ln)) for ln in K_LINES)
    kp_eng_off = all(absent(eng.get(ln)) for ln in KP_LINES)
    woa_missing = eng.get("DSV41_WOA_PREPACK=1 but this image")
    snap = [a.split("/snapshots/")[1] for a in json.load(open(f"{b}/four/serve_cmd.json")) if "/snapshots/" in a]
    v = dict(
        arm=arm, commit=P["commit"][:12],
        patch_sha256_list_digest=hashlib.sha256(open(f"{b}/patch.sha256", "rb").read()).hexdigest()[:16],
        snapshot=snap, k_envs_both_ranks_as_promoted=k_ok, kp_addons_off_both_ranks=kp_off,
        arm_envs_both_ranks=arm_state, arm_envs_as_arm=arm_ok,
        sparse_markov_line_head_worker={"any": sm_total, "k=256": k256, "k=1024": k1024}, sparse_markov_engagement_as_arm=sm_ok,
        k_lines_engaged_both_ranks=k_eng_ok, kp_lines_absent=kp_eng_off, woa_prepack_missing_warning=woa_missing,
        prose_long_c1=f"{P['prose_long_c1_finish']}, post-EOS {P['prose_long_c1_post_eos']}",
        prose_long_c2=f"{P['prose_long_c2_finish']}, post-EOS {P['prose_long_c2_post_eos']}",
        lail_finish=P["lail_finish"], serve_env_ranks_match=P["serve_env_ranks_match"],
        lever_disarmed=P["lever_disarmed"], disarm_scan_rc=P["disarm_scan_rc"], gpu_guard_rc=P["gpu_guard_rc"],
        contaminated=P["contaminated"], e2e_rc=P["e2e_rc"], correctness=P["correctness"],
        memavail_preconditions_spark1_spark2=mem_avail(f"{b}/preconditions.txt")[:2],
        memavail_after_smoke_spark1_spark2=P["memavail_after_smoke_gib"],
        four_numbers_partial=P["four_partial"],
    )
    v["valid"] = (k_ok and kp_off and arm_ok and sm_ok and k_eng_ok and kp_eng_off and absent(woa_missing)
                  and v["prose_long_c1"] == "length, post-EOS 0.0" and v["prose_long_c2"].startswith("length")
                  and v["lail_finish"] == ["length"] and v["serve_env_ranks_match"] and not v["lever_disarmed"]
                  and v["disarm_scan_rc"] == [0] and v["gpu_guard_rc"] == [0] and not v["contaminated"]
                  and v["e2e_rc"] == [0] and v["correctness"] == "8/8" and snap == ["2.0bpw-mcg-lmhead-mxfp8"]
                  and not v["four_numbers_partial"]
                  and all(int(x.rstrip("Gi")) >= 8 for x in v["memavail_after_smoke_spark1_spark2"]))

# ------------------------------------------------------------------ comparisons
def cmp(D, base, cand, key, lower_better, nd=3):
    """decision-bundle-compute.py.txt cmp() (same noise rule and fields), with a rounding digit count and the s9
    per-boot reading added: 'every S1024 boot beats the S median by more than the noise'."""
    xa, xb = [D[b][key] for b in BC.ARMS[base]], [D[b][key] for b in BC.ARMS[cand]]
    ma, mb = st.median(xa), st.median(xb)
    sa, sb = max(xa) - min(xa), max(xb) - min(xb)
    noise = max(sa, sb)
    imp = (ma - mb) if lower_better else (mb - ma)
    margins = [((ma - v) if lower_better else (v - ma)) for v in xb]
    return dict(base=[r(v, nd) for v in xa], cand=[r(v, nd) for v in xb], base_median=r(ma, nd), cand_median=r(mb, nd),
                base_spread=r(sa, nd), cand_spread=r(sb, nd), noise=r(noise, nd), improvement=r(imp, nd),
                improvement_pct=r(100.0 * imp / ma, 2), per_boot_margin_vs_base_median=[r(m, nd) for m in margins],
                every_cand_boot_beats_base_median=min(margins) > 0, beyond_noise=imp > noise,
                gate_pass=(imp > noise and min(margins) > 0), every_cand_boot_margin_gt_noise=min(margins) > noise,
                _raw=dict(ma=ma, mb=mb))


def overlap(D, base, cand, key):
    pa, pb = [D[b][key] for b in BC.ARMS[base]], [D[b][key] for b in BC.ARMS[cand]]
    return max(min(pa), min(pb)) <= min(max(pa), max(pb))


def cell_block(D, base, cand, cell):
    tk, mk, ak = CELLS[cell]
    tok, ms, acc = cmp(D, base, cand, tk, False), cmp(D, base, cand, mk, True), cmp(D, base, cand, ak, False, 4)
    def pct(c):  # candidate arm median vs base arm median, unrounded
        raw = c.pop("_raw")
        return 100.0 * (raw["mb"] / raw["ma"] - 1.0)
    acc_p, ms_p, tok_p = pct(acc), pct(ms), pct(tok)
    return dict(
        tok_s=tok, ms_step=ms, acceptance=acc,
        acceptance_per_boot_intervals_overlap=overlap(D, base, cand, ak),
        pct_cand_vs_base=dict(acceptance=r(acc_p, 2), ms_step=r(ms_p, 2), tok_s=r(tok_p, 2),
                              tok_s_from_acc_and_ms=r(100.0 * ((1 + acc_p / 100) / (1 + ms_p / 100) - 1), 2),
                              note="arm medians; tok/s ~ tokens per step / ms per step, so the last value is the acceptance change net of the ms/step change"),
    )


def comparison(D, base, cand):
    return {cell: cell_block(D, base, cand, cell) for cell in CELLS}


def gate(comp):
    g = {}
    for cell in GATE:
        t = comp[cell]["tok_s"]
        g[cell + "_tok_s"] = dict(base=t["base"], cand=t["cand"], base_median=t["base_median"], cand_median=t["cand_median"],
                                  improvement=t["improvement"], noise=t["noise"],
                                  per_boot_margin_vs_base_median=t["per_boot_margin_vs_base_median"],
                                  armsmd_rule_median_imp_gt_noise_and_every_boot_beats_median=t["gate_pass"],
                                  s9_rule_every_boot_margin_gt_noise=t["every_cand_boot_margin_gt_noise"] and t["every_cand_boot_beats_base_median"])
    return g


def noninf(D, base, cand):
    return {k: BC.noninf(D, base, cand, k) for k in NI_KEYS}


# ------------------------------------------------------------------ robustness context (not the gate)
def rank_ctx(D, pop_boots, sub, key, higher_better):
    pop = {b: (-D[b][key] if higher_better else D[b][key]) for b in pop_boots + sub}
    res = AD.exact_rank_sum_low(pop, sub)
    order = sorted(pop, key=lambda b: pop[b])
    res["rank_of_sub_best_first"] = [order.index(b) + 1 for b in sub]
    res["direction"] = "best = highest" if higher_better else "best = lowest"
    return res


def mwu_runs(D, xs, ys, key):
    return AD.mwu([v for b in xs for v in D[b][key]], [v for b in ys for v in D[b][key]])


def selfcons(b):
    return AD.selfcons_runs(b)

                + [AD.env_of(b, n).get("DSV41_DSPARK_SPARSE_MARKOV_TOPK") for n in (1, 2)] for b in KCFG}

    c1 = comparison(D, "S", "W")          # counted pair 1: cand W (revert) vs base S (incumbent default)
    c1_s = comparison(D, "W", "S")        # same boots, S as the candidate: S's ms/step gain vs its acceptance cost

    s1024_scfg = comparison(D, "Scfg10", "S1024")

    ranks = {}
    for name, sub in (("W", W), ("S1024", S1024)):
        for cell in GATE:
            tk, mk, ak = CELLS[cell]
            ranks[f"{name}_in_Scfg10_{cell}_tok_s"] = rank_ctx(D, SCFG, sub, tk, True)
            ranks[f"{name}_in_Scfg10_{cell}_acceptance"] = rank_ctx(D, SCFG, sub, ak, True)
            ranks[f"{name}_in_Scfg10_{cell}_ms_step"] = rank_ctx(D, SCFG, sub, mk, False)
    runs = dict(
        note="per-run Mann-Whitney U, y = the candidate arm; L.A.I.L 10 runs per boot, prose_long c=1 5 runs, four prose c=1 9 runs",
        W_vs_S_lail_tok=mwu_runs(D, S12, W, "lail_tok_runs_full"),

        S1024_vs_S34_lail_acc=mwu_runs(D, S34, S1024, "acc_lail_runs"),
        S1024_vs_S34_prose_long_c1_acc=mwu_runs(D, S34, S1024, "acc_prose_long_c1_runs"),
        S1024_vs_S34_four_prose_c1_acc=mwu_runs(D, S34, S1024, "acc_four_prose_c1_runs"),
        S1024_vs_Scfg10_lail_tok=mwu_runs(D, SCFG, S1024, "lail_tok_runs_full"),
        S1024_vs_Scfg10_lail_acc=mwu_runs(D, SCFG, S1024, "acc_lail_runs"),
        S1024_vs_Scfg10_prose_long_c1_acc=mwu_runs(D, SCFG, S1024, "acc_prose_long_c1_runs"),
        W_vs_Scfg10_lail_acc=mwu_runs(D, SCFG, W, "acc_lail_runs"),
        W_vs_Scfg10_prose_long_c1_acc=mwu_runs(D, SCFG, W, "acc_prose_long_c1_runs"),
        S1024_vs_S34_pp_warm_8k=AD.mwu([v for b in S34 for v in D[b]["pp_runs"]["pp_warm_8k"]],
                                       [v for b in S1024 for v in D[b]["pp_runs"]["pp_warm_8k"]]),
        S1024_vs_Scfg10_pp_warm_8k=AD.mwu([v for b in SCFG for v in D[b]["pp_runs"]["pp_warm_8k"]],

        pairs = []

            for rx in sc[x]:
                for ry in sc[y]:
                    pairs += list(zip(rx, ry))
        return AD.hazard(pairs)
    greedy = dict(
        note="quality_quick selfcons: 12 greedy prompts x 128 tokens, 2 runs per boot. Hazard = first-divergence events / tokens at risk, pooled over every run pair of the named boots (decision-addons-compute.py.txt hazard()). If an arm changed the greedy output, arm-vs-S would diverge more than S-vs-S across boots.",

    )

def slim(comp):
    keep = ("base", "cand", "base_median", "cand_median", "base_spread", "cand_spread", "noise", "improvement",
            "improvement_pct", "per_boot_margin_vs_base_median", "every_cand_boot_beats_base_median", "beyond_noise",
            "gate_pass", "every_cand_boot_margin_gt_noise")
    out = {}
    for cell, blk in comp.items():
        out[cell] = {m: {k: blk[m][k] for k in keep if k in blk[m]} for m in ("tok_s", "ms_step", "acceptance")}
        out[cell]["acceptance_per_boot_intervals_overlap"] = blk["acceptance_per_boot_intervals_overlap"]
        out[cell]["pct_cand_vs_base"] = blk["pct_cand_vs_base"]
    return out


def pr(v):
    return "/".join(f"{x:g}" for x in v)


def compose(c):
    D, val = c["D"], c["val"]
    g1, g2 = gate(c["c1"]), gate(c["c2"])
    A = "armsmd_rule_median_imp_gt_noise_and_every_boot_beats_median"
    B = "s9_rule_every_boot_margin_gt_noise"
    valid_all = all(v["valid"] for v in val.values())
    q_all = all(q["passed"] for q in c["qual"].values())
    ni2_ok = all(v["ok_vs_larger_spread"] for v in c["ni2"].values())
    revert_both = all(g1[k][A] for k in g1)
    revert_any = any(g1[k][A] for k in g1)
