def sign_p(k, n):
    """Exact two-sided binomial(n, 0.5) p-value for k successes (sign test / exact McNemar)."""
    if n == 0:
        return 1.0
    lo = min(k, n - k)
    return min(1.0, 2 * sum(math.comb(n, i) for i in range(lo + 1)) / 2 ** n)


def wilcoxon_exact(d):
    """Exact two-sided Wilcoxon signed-rank p (zeros dropped, ties by average rank, exact
    conditional distribution over doubled ranks)."""
    d = [x for x in d if x != 0]
    n = len(d)
    order = sorted(range(n), key=lambda i: abs(d[i]))
    r2 = [0] * n
    i = 0
    while i < n:
        j = i
        while j + 1 < n and abs(d[order[j + 1]]) == abs(d[order[i]]):
            j += 1
        for k in range(i, j + 1):
            r2[order[k]] = (i + 1) + (j + 1)
        i = j + 1
    dist = {0: 1}
    for rr in r2:
        nd = dict(dist)
        for s, c in dist.items():
            nd[s + rr] = nd.get(s + rr, 0) + c
        dist = nd
    wp = sum(r2[i] for i in range(n) if d[i] > 0)
    tot = 2 ** n
    lo = sum(c for s, c in dist.items() if s <= wp) / tot
    hi = sum(c for s, c in dist.items() if s >= wp) / tot
    return dict(n_nonzero=n, w_plus=wp / 2, w_minus=(sum(r2) - wp) / 2, p_two_sided=min(1.0, 2 * min(lo, hi)))


def boot_ci_mean(d, reps=20000, seed=20260926):
    rng = random.Random(seed)
    n = len(d)
    ms = sorted(sum(rng.choice(d) for _ in range(n)) / n for _ in range(reps))
    return [ms[int(0.025 * reps)], ms[int(0.975 * reps) - 1]]


def mwu(x, y):
    """Mann-Whitney U of y vs x, normal approximation with tie and continuity correction."""
    n1, n2 = len(x), len(y)
    allv = sorted(x + y)
    rank, i = {}, 0
    while i < len(allv):
        j = i
        while j + 1 < len(allv) and allv[j + 1] == allv[i]:
            j += 1
        rank[allv[i]] = (i + 1 + j + 1) / 2
        i = j + 1
    u2 = sum(rank[v] for v in y) - n2 * (n2 + 1) / 2
    mu, N = n1 * n2 / 2, n1 + n2
    tie = sum(t ** 3 - t for t in collections.Counter(allv).values())
    var = n1 * n2 / 12 * ((N + 1) - tie / (N * (N - 1)))
    z = 0.0 if u2 == mu else (u2 - mu - math.copysign(0.5, u2 - mu)) / math.sqrt(var)
    return dict(n_cur=n1, n_vit=n2, U_vit=u2, z=r(z, 3), p_two_sided_normal=r(math.erfc(abs(z) / math.sqrt(2)), 5),
                hodges_lehmann_vit_minus_cur=r(st.median([b - a for a in x for b in y]), 4))


# ------------------------------------------------------------------ speed (ARMS.md step 6)
def cmp(data, key, lower_better):
    xa, xb = [data[b][key] for b in CUR], [data[b][key] for b in VIT]
    ma, mb = st.median(xa), st.median(xb)
    sa, sb = max(xa) - min(xa), max(xb) - min(xb)
    noise = max(sa, sb)
    imp = (ma - mb) if lower_better else (mb - ma)
    better_all = all((v < ma) if lower_better else (v > ma) for v in xb)
    worse_all = all((v > ma) if lower_better else (v < ma) for v in xb)
    return dict(CUR=[r(v) for v in xa], VIT=[r(v) for v in xb], CUR_median=r(ma), VIT_median=r(mb),
                delta_VIT_minus_CUR=r(mb - ma), delta_pct=r(100.0 * (mb - ma) / ma, 2),
                CUR_spread=r(sa), VIT_spread=r(sb), noise=r(noise), improvement=r(imp),
                every_VIT_boot_beats_CUR_median=better_all, every_VIT_boot_worse_than_CUR_median=worse_all,
                better_beyond_noise=(imp > noise and better_all), worse_beyond_noise=((-imp) > noise),
                within_noise=abs(imp) <= noise)


def acc_overlap(per_boot_key, runs_of):
    pa, pb = [D[b][per_boot_key] for b in CUR], [D[b][per_boot_key] for b in VIT]
    ra = [v for b in CUR for v in runs_of(b)]
    rb = [v for b in VIT for v in runs_of(b)]
    return dict(CUR_per_boot=[r(v, 4) for v in pa], VIT_per_boot=[r(v, 4) for v in pb],
                per_boot_interval_overlap=max(min(pa), min(pb)) <= min(max(pa), max(pb)),
                CUR_run_range=[r(min(ra), 4), r(max(ra), 4)], VIT_run_range=[r(min(rb), 4), r(max(rb), 4)],
                run_range_overlap=max(min(ra), min(rb)) <= min(max(ra), max(rb)),
                mwu_runs=mwu(ra, rb))

def paired_binary(name, universe, ok, pairs, aa_pairs):
    """ok[(boot, mode)] = set of correct items. Exact McNemar per interleaved pair, pooled
    per-item sign test over all readings of each arm, and the same-arm A/A discordance."""
    per_pair = {}
    for c, v in pairs:
        b_ = len(ok[c] - ok[v])          # CUR right, VIT wrong
        c_ = len(ok[v] - ok[c])          # CUR wrong, VIT right
        per_pair[f"{v[0]}/{v[1]} vs {c[0]}/{c[1]}"] = dict(
            CUR_k=len(ok[c]), VIT_k=len(ok[v]), n=len(universe), CUR_only=b_, VIT_only=c_,
            CUR_only_items=sorted(ok[c] - ok[v]), VIT_only_items=sorted(ok[v] - ok[c]),
            mcnemar_exact_p=r(sign_p(c_, b_ + c_), 5))
    cur_keys = [c for c, _ in pairs]
    vit_keys = [v for _, v in pairs]
    score = {i: sum(i in ok[v] for v in vit_keys) - sum(i in ok[c] for c in cur_keys) for i in universe}
    pos = sorted(i for i, s in score.items() if s > 0)
    neg = sorted(i for i, s in score.items() if s < 0)
    aa = {}
    for x, y in aa_pairs:
        aa[f"{y[0]}/{y[1]} vs {x[0]}/{x[1]}"] = dict(discordant=len(ok[x] ^ ok[y]), items=sorted(ok[x] ^ ok[y]))
    return dict(component=name, n_items=len(universe), per_pair=per_pair,
                pooled=dict(readings_per_arm=len(pairs), items_VIT_better=len(pos), items_CUR_better=len(neg),
                            VIT_better_items=pos, CUR_better_items=neg,
                            sign_test_exact_p=r(sign_p(len(pos), len(pos) + len(neg)), 5),
                            CUR_total_correct=sum(len(ok[c]) for c in cur_keys),
                            VIT_total_correct=sum(len(ok[v]) for v in vit_keys),
                            total_items_scored_per_arm=len(universe) * len(pairs)),
                same_arm_AA_discordance=aa)


def first_div(a, b):
    for i, (x, y) in enumerate(zip(a, b)):
        if x != y:
            return i
    return None if len(a) == len(b) else min(len(a), len(b))


def hazard(pairs):
    ev, risk, divs = 0, 0, []
    for a, b in pairs:
        d = first_div(a, b)
        divs.append(d)
        if d is None:
            risk += len(a)
        else:
            ev += 1
            risk += d + 1
    return dict(hazard=r(ev / risk, 5) if risk else 0.0, diverged=ev, pairs=len(pairs), first_div=divs)


def vs_golden(target_runs, golden_run0):
    return hazard([(s, golden_run0[i]) for run in target_runs for i, s in enumerate(run)])


def bytes_to_unicode():
    bs = list(range(ord("!"), ord("~") + 1)) + list(range(ord("¡"), ord("¬") + 1)) + list(range(ord("®"), ord("ÿ") + 1))
    cs = bs[:]
    n = 0
    for b in range(256):
        if b not in bs:
            bs.append(b)
            cs.append(256 + n)
            n += 1
    return dict(zip(bs, [chr(c) for c in cs]))


def make_decoder():
    if not os.path.exists(TOK):
        return None
    t = json.load(open(TOK))
    inv = {v: k for k, v in t["model"]["vocab"].items()}
    added = {a["id"]: a["content"] for a in t["added_tokens"]}
    u2b = {v: k for k, v in bytes_to_unicode().items()}

    def dec(ids):
        out, buf = [], bytearray()
        for i in ids:
            if i in added:
                out.append(buf.decode("utf-8", "replace"))
                buf = bytearray()
                out.append(added[i])
            else:
                buf.extend(u2b[c] for c in inv.get(i, "") if c in u2b)
        out.append(buf.decode("utf-8", "replace"))
        return "".join(out)
    return dec

                           ("median_abs_dlogprob", "p99_abs_dlogprob", "mean_abs_dlogprob", "gen_nll_prefill", "gen_nll_decode")}
              for b in ORDER for m in MODES}

    # ---------------- quality: greedy self-consistency control

    base_runs = {m: BASE[m]["components"]["selfcons"]["runs"] for m in MODES}

    sc["recomputed_equals_stored"] = recompute_ok
    cross = {}
    for m in MODES:
        cross[f"CUR-2/{m} vs CUR-1/{m} run A (same pack, cross-boot)"] = vs_golden(runs[("CUR-2", m)], runs[("CUR-1", m)][0])
        cross[f"VIT-2/{m} vs VIT-1/{m} run A (same pack, cross-boot)"] = vs_golden(runs[("VIT-2", m)], runs[("VIT-1", m)][0])

    out["speed"] = speed

    doc = decide(out)
    json.dump(doc, open(sys.argv[1], "w"), indent=1, default=str)

    return f"{r(min(xs), n)}-{r(max(xs), n)}"
