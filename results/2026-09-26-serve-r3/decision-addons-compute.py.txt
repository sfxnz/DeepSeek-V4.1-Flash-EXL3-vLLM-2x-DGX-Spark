# ------------------------------------------------------------------ statistics (stdlib)
def avg_ranks(vals):
    s = sorted(vals)
    rk, i = {}, 0
    while i < len(s):
        j = i
        while j + 1 < len(s) and s[j + 1] == s[i]:
            j += 1
        rk[s[i]] = (i + 1 + j + 1) / 2
        i = j + 1
    return rk


def mwu(x, y):
    """Mann-Whitney U of y vs x, normal approximation with tie and continuity correction."""
    n1, n2 = len(x), len(y)
    rk = avg_ranks(x + y)
    u2 = sum(rk[v] for v in y) - n2 * (n2 + 1) / 2
    mu, N = n1 * n2 / 2, n1 + n2
    tie = sum(t ** 3 - t for t in [sum(1 for v in x + y if v == u) for u in set(x + y)])
    var = n1 * n2 / 12 * ((N + 1) - tie / (N * (N - 1)))
    z = 0.0 if u2 == mu else (u2 - mu - math.copysign(0.5, u2 - mu)) / math.sqrt(var)
    return dict(n_x=n1, n_y=n2, U_y=u2, z=r(z, 3), p_two_sided_normal=r(math.erfc(abs(z) / math.sqrt(2)), 5),
                hodges_lehmann_y_minus_x=r(st.median([b - a for a in x for b in y]), 4))


def exact_rank_sum_low(pop, sub):
    """Exact one-sided p that `sub` (a set of labels in pop) has a rank sum this low or lower among all
    same-size subsets of pop (average ranks for ties)."""
    vals = list(pop.values())
    rk = avg_ranks(vals)
    rsum = sum(rk[pop[k]] for k in sub)
    keys = list(pop)
    combos = list(itertools.combinations(keys, len(sub)))
    le = sum(1 for c in combos if sum(rk[pop[k]] for k in c) <= rsum + 1e-9)
    return dict(n_pop=len(pop), n_sub=len(sub), rank_sum=rsum, p_one_sided_low=r(le / len(combos), 4),
                subsets=len(combos))


def first_div(a, b):
    for i, (x, y) in enumerate(zip(a, b)):
        if x != y:
            return i
    return None if len(a) == len(b) else min(len(a), len(b))


def hazard(pairs):
    ev, risk = 0, 0
    for a, b in pairs:
        d = first_div(a, b)
        if d is None:
            risk += len(a)
        else:
            ev += 1
            risk += d + 1
    return dict(hazard=r(ev / risk, 5) if risk else 0.0, diverged=ev, pairs=len(pairs))


# ------------------------------------------------------------------ per-boot readers (in addition to BC.boot)
def env_of(b, node):
    return dict(x.split("=", 1) for x in json.load(open(f"{b}/four/serve_env_spark{node}.json")) if "=" in x)


def engagement(b):
    out = {}
    for line in open(f"{b}/engagement.txt"):
        m = re.match(r"^(.+?)\s+'([^']*)'\s+head\(spark1\)=(\d+)\s+worker\(spark2\)=(\d+)", line)
        if m:
            out[m.group(2)] = [int(m.group(3)), int(m.group(4))]
    return out


def mem_avail(path):
    return [line.split()[6] for line in open(path) if line.startswith("Mem:")]


def model_loading(b):
    ts = AUDIT_TS.get(b)
    out = {}
    for role in ("head", "worker"):
        p = os.path.join(W, ".run-state", f"audit-{role}-20260926-{ts}.log") if ts else None
        if p and os.path.exists(p):
            m = re.search(r"Model loading took ([\d.]+) GiB memory and ([\d.]+) seconds", open(p, errors="replace").read())
            out[role] = dict(file=os.path.relpath(p, W), GiB=float(m.group(1)), s=r(float(m.group(2)), 2)) if m else None
    return out


def lail_acc_runs(b):
    return [json.loads(m.group(1))["acceptance_len"]
            for m in (re.match(r"run=\d+ (\{.*\})$", l.strip()) for l in open(f"{b}/lail10.txt")) if m]


def bench_runs(b, phase, conc):
    out = []
    for l in open(f"{b}/bench_decode.txt"):
        m = re.match(rf"phase={phase} c={conc} run=\d+ .* per_stream=\[([^\]]*)\] ttft=\[([^\]]*)\] acc=([\d.]+) ms_step=\[([^\]]*)\]", l)
        if m:
            out.append(dict(tok_s=[float(x) for x in m.group(1).split(",")], acc=float(m.group(3)),
                            ms_step=[float(x) for x in m.group(4).split(",")]))
    return out


def correctness_outputs(b):
    rows = {}
    for l in open(f"{b}/correctness.txt"):
        m = re.match(r"^(PASS|FAIL) (\S+) \([\d.]+s\) (.*)$", l.strip())
        if m:
            rows[m.group(2)] = (m.group(1), m.group(3))
    return rows


def selfcons_runs(b):
